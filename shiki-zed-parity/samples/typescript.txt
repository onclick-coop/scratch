/** Retries a failed fetch before giving up. */
const MAX_RETRIES = 3
namespace Limits { export const TIMEOUT = 30 }
export enum Status { Active = 'active', Archived }

@Injectable()
export class UserService<T extends object> implements Store {
  private readonly cache = new Map<string, T>()
  constructor(private readonly db: Database, public name?: string) {
    super()
    this.name = name ?? `user-${db.id}`
  }

  async fetch({ id, limit = 10 }: Query, [first]: string[]): Promise<T | null> {
    const pattern = /^\d+$/g
    if (!pattern.test(id)) throw new Error(`bad id: ${id}\n`)
    return this.cache.get(id) ?? null
  }
}
